/* ---------- tokens (HMAC-signed, short-lived) + password hashing ---------- */
const ctEq_ = (a, b) => { if (a.length !== b.length) return false; let r = 0; for (let i = 0; i < a.length; i++) r |= a.charCodeAt(i) ^ b.charCodeAt(i); return r === 0; };

const sign_ = o => { const body = b64_(JSON.stringify(o)); return body + '.' + b64_(Utilities.computeHmacSha256Signature(body, SECRET)); };
function verify_(t) {
  if (typeof t !== 'string' || t.indexOf('.') < 0) return null; const parts = t.split('.');
  if (!ctEq_(parts[1], b64_(Utilities.computeHmacSha256Signature(parts[0], SECRET)))) return null;
  const o = JSON.parse(Utilities.newBlob(Utilities.base64DecodeWebSafe(parts[0])).getDataAsString()); return o.exp > Date.now() / 1000 ? o : null;
}
function hashPw_(pw, saltHex) {   // PBKDF2-HMAC-SHA256 (one 32-byte block) + pepper
  const key = Utilities.newBlob(pw + PEPPER).getBytes(), salt = Utilities.newBlob(Utilities.base64Decode(saltHex)).getBytes().concat([0, 0, 0, 1]);
  let u = Utilities.computeHmacSha256Signature(salt, key), t = u.slice();
  for (let i = 1; i < ITER; i++) { u = Utilities.computeHmacSha256Signature(u, key); for (let j = 0; j < 32; j++) t[j] ^= u[j]; }
  return Utilities.base64Encode(t);
}
const newSalt_ = () => Utilities.base64Encode(Utilities.newBlob(Utilities.getUuid() + Utilities.getUuid()).getBytes().slice(0, 16));

}

function manageEvent(data, action) {   // 'add' | 'edit' | 'cancel'
  const from = data.eventDateFrom, to = data.eventDateTo || from;
  if (!from || new Date(to) < new Date(from)) return { ok: false, code: 'VALIDATION', errors: ['eventDateTo'] };
  const cal = CalendarApp.getCalendarById(P.getProperty('CAL_ID'));
  const days = []; for (let d = new Date(from); d <= new Date(to); d.setDate(d.getDate() + 1)) days.push(Utilities.formatDate(d, 'GMT+4', 'yyyy-MM-dd'));
  if (days.length > 60) return { ok: false, code: 'VALIDATION', errors: ['eventDateTo'] };
  if (action === 'cancel') {   // إلغاء الفعالية: تحرير كل حجوزات النطاق + حذف أحداث التقويم تلقائياً
    const rows = table_('الفعاليات_القادمة'), idI = rows.h.indexOf('المعرف');
    for (let r = 0; r < rows.rows.length; r++) if (String(rows.rows[r][idI]) === String(data.id)) rows.sh.getRange(r + 2, 1, 1, rows.h.length).setValues([rows.rows[r].map((v, j) => j === rows.h.indexOf('الحالة') ? 'ملغاة' : v)]);
    table_('حجز_القاعات').sh.getDataRange().getValues().forEach((row, idx) => { const i2 = {}; table_('حجز_القاعات').h.forEach((h, j) => i2[h] = j); if (String(row[i2['الرقم الوظيفي']]) === String(data.jobId) && days.indexOf(String(row[i2['التاريخ']])) > -1) { const ev = cal.getEventById(String(row[i2['معرّف الحدث']])); if (ev) ev.deleteEvent(); } });
    audit_(data.jobId, 'EVENT_CANCELLED', data.id + ' (' + from + '→' + to + ')'); return { ok: true };
  }
  for (const di of days) {   // فحص تعارض يومياً عبر النطاق: حجوزات + تقويم (All-Day = 08:00–13:00) + الفترات المحمية
    const dow = new Date(di).getDay(); if (dow === 5 || dow === 6) return { ok: false, code: 'VALIDATION', errors: ['eventDateFrom'] };
    const bad = txt => ({ ok: false, code: 'RANGE_CONFLICT', message: 'تعارض في ' + di + ' (' + dayName_(dow) + ') • ' + txt + ' / Conflict on ' + di + ' • ' + txt });
    const sh = table_('حجز_القاعات'); const ci = sh.h.indexOf('التاريخ'), ri = sh.h.indexOf('القاعة');
    for (const r of sh.rows) if (String(r[ci]) === di && String(r[ri]) === String(data.room)) return bad('محجوز بالفعل / already booked');
    const pr = row_('ProtectedSlots', 'day', String(dow)); if (pr && pr.s < data.timeTo && data.timeFrom < pr.e) return bad('فترة محمية ' + pr.label + ' / protected slot ' + pr.label);
    for (const ev of cal.getEventsForDay(new Date(di))) {
      const st = ev.getAllDayEvent() ? new Date(di + 'T08:00:00+04:00') : ev.getStartTime(), en = ev.getAllDayEvent() ? new Date(di + 'T13:00:00+04:00') : ev.getEndTime();
      if (st < new Date(di + 'T' + data.timeTo + ':00+04:00') && new Date(di + 'T' + data.timeFrom + ':00+04:00') < en) return bad('حدث التقويم: ' + ev.getTitle() + ' / calendar: ' + ev.getTitle());
    }
  }

  }
  const srow = table_('الفعاليات_القادمة');
  const rec = [data.id || ('EV' + new Date().getTime()), data.titleAr, data.titleEn, data.aud, from, to, data.timeFrom, data.timeTo, data.room, evIds.join(','), evIds.join(','), data.dept, data.approve ? 'نعم' : 'لا', data.cap, data.kind, data.desc || '', 'معتمد ومنشور', new Date()];
  if (action === 'edit') { const idI = srow.h.indexOf('المعرف'); for (let r = 0; r < srow.rows.length; r++) if (String(srow.rows[r][idI]) === String(data.id)) { srow.sh.getRange(r + 2, 1, 1, srow.h.length).setValues([rec]); break; } }

  return { ok: true, eventDateFrom: from, eventDateTo: to, eventIds: evIds };
}
